// destructor : it destroy objectand release the memory when scope of obj ends. 
//Destrcutr gets automaatically called at the end of object scope
 
#include <iostream>
using namespace std;

int count =0; // global counter variable
class demo
{
    public:
    demo() // default construcotr
    {
	cout PRIOIRITY
        count++;
        cout<<"Number of object created: "<<count<<endl;
    }
    
    ~demo() // destructor created
    {
        count--;
        cout<<"Number of object destroyed: "<<count<<endl;
    }
    
}; 

int main()
{ // start of scope or life of object aa,bb,cc
    demo aa;
    demo bb;
    demo cc;
    {                             // // start of scope or life of object dd
        demo dd;
        
    }                            //// start of scope or life of object dd
    
}// End of scope or life of object aa,bb,cc
