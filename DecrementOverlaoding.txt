#WAP FOR PREFIX AND POSFIX DECREMENT, USING FUNCTION OVERLOADING
#include <iostream>

using namespace std;
class demo
{
    int x;
    public:
    void getdata()
    {
        cout<<"ENter x= "<<endl;
        cin>>x;
    }
    
    void putdata()
    {
        cout<<"X is :"<<x<<endl;
    }

    // HERE DONE FUNCTION OVERLOADING
    void operator--()
    {
       
        x = x-1;
        cout<<"I am prefix called: x is :"<<x<<endl;
    }
    void operator-- (int)
    {
        
        x = x-1;
        cout<<"I am overloaded function for postfix: x is : "<<x<<endl;
    }
};

int main()
{
 demo aa;
 aa.getdata();
 aa.putdata();
 
 --aa; //prrfix decrement
 aa--; //postfix decerement
 aa.putdata();
}
