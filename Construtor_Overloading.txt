// CNSTRUCTOR OVERLOADING

#include <iostream>

using namespace std;
class demo
{
    int a;
    public:
    
    demo()
    {
       a=10; 
    }
    demo(int x)
    {
        a =x;
    }
    demo(demo &z)
    {
        a = z.a;
    }
    void display()
    {
        cout<<"A is:"<<a<<endl;
    }
};
int main()
{
    demo aa; //a =20
    aa.display();
    
    demo bb(20);//a=20
    bb.display();
    
    demo cc(bb); //a=20
    cc.display();
}
